-- ============================================ -- Estoque mínimo por produto (ponto de reposição) -- Rode este script no SQL Editor do Supabase (é seguro, não apaga dados) -- ============================================ -- Cada produto passa a ter seu próprio limite de estoque baixo. -- O padrão é 5 pra manter o comportamento atual nos produtos já cadastrados -- (você pode ajustar cada um depois, editando o produto no sistema). alter table products add column if not exists estoque_minimo numeric not null default 5; -- Atualiza a view de saldo pra já trazer esse valor junto -- (adicionado no final da lista de colunas — não precisa apagar a view) create or replace view product_balances with (security_invoker = true) as select p.codigo, p.produto, p.categoria, p.unidade, p.codigo_barras, p.fornecedor_id, f.nome as fornecedor_nome, coalesce(sum(case when m.tipo = 'entrada' then m.quantidade else 0 end), 0) as total_entrada, coalesce(sum(case when m.tipo = 'saida' then m.quantidade else 0 end), 0) as total_saida, coalesce(sum(case when m.tipo = 'entrada' then m.quantidade else 0 end), 0) - coalesce(sum(case when m.tipo = 'saida' then m.quantidade else 0 end), 0) as saldo, case when coalesce(sum(case when m.tipo = 'entrada' and m.origem = 'manual' then m.quantidade else 0 end), 0) > 0 then round( coalesce(sum(case when m.tipo = 'entrada' and m.origem = 'manual' then m.quantidade * coalesce(m.valor, 0) else 0 end), 0) / sum(case when m.tipo = 'entrada' and m.origem = 'manual' then m.quantidade else 0 end) , 4) else 0 end as custo_medio, p.estoque_minimo from products p left join movements m on m.codigo = p.codigo left join fornecedores f on f.id = p.fornecedor_id group by p.codigo, p.produto, p.categoria, p.unidade, p.codigo_barras, p.fornecedor_id, f.nome, p.estoque_minimo; grant select on product_balances to authenticated; -- ============================================ -- Pronto. Depois de rodar isso, publique o novo index.html. -- No cadastro/edição de cada produto, agora dá pra definir o -- estoque mínimo dele — o alerta de "estoque baixo" passa a -- respeitar esse valor específico, em vez de um número fixo pra todos. -- ============================================
